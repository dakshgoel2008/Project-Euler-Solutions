void solve() {
    vector<pair<ll,ll>>a;
    for(ll i=1;i<=1000000;++i){
        ll cnt=0;
        ll n=i;
        while(n>1){
            if(n&1)n=3*n+1;
            else n/=2;
            cnt++;
        }
        a.push_back({cnt,i});
    }
    sort(a.begin(),a.end());
    cout<<a.back().second<<endl;
}
