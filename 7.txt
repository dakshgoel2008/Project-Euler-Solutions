void solve(){
    int n=200010;
    vector<ll> spf(n+1);
    iota(all(spf),0);
    for(ll i=2;i*i<=n;++i){
        if(spf[i]==i){
            for(ll j=i*i;j<=n;j+=i){
                if(spf[j]==j)spf[j]=i;
            }
        }
    }
    ll cnt=0;
    ll ans=0;
    for(ll i=2;i<=n;++i){
        if(spf[i]==i){
            cnt++;
            if(cnt==10001){
                ans=i;
                break;
            }
        }
    }
    cout<<ans<<endl;
}
