void solve() {
    string s;
    cin >> s;
    ll ans = 0;
    for (int i=0; i+13<=s.size();++i) {
        ll prod=1;
        for (int j=i;j<i+13;++j) {
            prod*=(s[j]-'0');
        }
        ans=max(ans,prod);
    }

    cout<<ans<<endl;
}
