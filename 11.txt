    int a[20][20];
    fr(i,20){
        fr(j,20){

        }
    }
    ll maxi=0;
    fr(i,20){
        fr(j,20){
            fr(d,8){
                ll ans=a[i][j];
                bool hai=true;
                for(int k=1;k<=3;++k){
                    int ni=i+(dx[d]*k);
                    int nj=j+(dy[d]*k);
                    if(ni<20&&nj<20&&ni>=0&&nj>=0){
                        ans*=a[ni][nj];
                    }else {

                        break;
                    }
                }
                if(hai)maxi=max(ans,maxi);
            }
        }
    }
    cout<<maxi<<endl;
}
