void solve() {
    int ans=0;
    for(int a=1;a<=1000;++a){
        for(int b=1;b<=1000;++b){
            for(int c=1;c<=1000;++c){
                int sum=a+b+c;
                if(sum==1000){
                    if((a*a+b*b)==c*c){
                        ans=a*b*c;
                        cout<<ans<<endl;
                        return;
                    }
                }
            }
        }
    }
}
