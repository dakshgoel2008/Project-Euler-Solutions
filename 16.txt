void solve(){
    vector<ll>d;
    d.pb(1);
    for(int i=0;i<1000;++i){
        int c=0;
        for(int j=0;j<d.size();++j){
            ll p=d[j]*2+c;
            d[j]=p%10;
            c=p/10;
        }
        while(c>0){
            d.pb(c%10); c/=10;
        }
    }
    ll sum=accumulate(all(d),0LL);
    cout<<sum<<endl;
}
